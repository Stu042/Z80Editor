cmake_minimum_required(VERSION 3.16)
project(Z80Editor VERSION 1.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Qt6 REQUIRED COMPONENTS Widgets)
include(CTest)
set(Z80TEST_SOURCE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../Z80Test" CACHE PATH "Path to the Z80Test source project")
if(NOT EXISTS "${Z80TEST_SOURCE_DIR}/CMakeLists.txt")
    message(FATAL_ERROR "Set Z80TEST_SOURCE_DIR to the existing Z80Test project directory")
endif()
function(add_simulator)
    # The editor runs the CLI; its own tests exercise that integration.
    set(BUILD_TESTING OFF)
    add_subdirectory("${Z80TEST_SOURCE_DIR}" simulator EXCLUDE_FROM_ALL)
endfunction()
add_simulator()
add_library(debug_core STATIC "${Z80TEST_SOURCE_DIR}/src/assembler.c" "${Z80TEST_SOURCE_DIR}/src/cpu.c")
target_include_directories(debug_core PUBLIC "${Z80TEST_SOURCE_DIR}/src")
set(CMAKE_AUTOMOC ON)
add_library(editor_ui src/editor.cpp src/editor.h src/window.cpp src/window.h src/debugsession.cpp src/debugsession.h)
target_link_libraries(editor_ui PUBLIC Qt6::Widgets debug_core)
target_include_directories(editor_ui PUBLIC src)
target_compile_definitions(editor_ui PRIVATE Z80TEST_DEFAULT_EXECUTABLE="$<TARGET_FILE:Z80Test>")
add_dependencies(editor_ui Z80Test)
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
    target_compile_options(editor_ui PRIVATE -Wall -Wextra -Wpedantic)
endif()
add_executable(Z80Editor src/main.cpp)
target_link_libraries(Z80Editor PRIVATE editor_ui)
if(BUILD_TESTING)
    find_package(Qt6 REQUIRED COMPONENTS Test)
    add_executable(editor_tests tests/editor_tests.cpp)
    target_link_libraries(editor_tests PRIVATE editor_ui Qt6::Test)
    add_test(NAME editor_tests COMMAND editor_tests)
    set_tests_properties(editor_tests PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
endif()
